/* ============================================================================
   Migration 017 - N° de change sur les HNO
   Demande client du 06/11/2026 : une HNO porte obligatoirement son numéro de
   change ; il apparaît dans la colonne « HNO - Change » des tableaux de
   facturation (« Copier pour e-mail »).

   Stockage : colonne existante saisie_activite.change_ou_motif (déjà utilisée
   par l'astreinte, NVARCHAR(100)). Aucune nouvelle colonne, aucune donnée
   modifiée. Seule la contrainte CK_saisie_astreinte_champs change :
     - ASTREINTE : inchangé (change obligatoire, résolu Oui/Non obligatoire) ;
     - HNO       : change_ou_motif AUTORISÉ, champs de résolution interdits ;
     - NORMALE   : inchangé (aucun champ d'astreinte).
   L'obligation du n° de change sur une HNO est portée par le serveur
   (Cra.Regles.dll du 06/11/2026) au moment de l'enregistrement : les HNO déjà
   en base sans numéro restent valides et le numéro est demandé dès que leur
   journée est modifiée.

   ORDRE DE MISE EN PRODUCTION : 017 -> Cra.Regles.dll -> index.html.
   Idempotent (réexécutable sans dégât).
     sqlcmd -S .\SQLEXPRESS -d cra -b -i 017_hno_numero_change.sql

   Retour arrière (seulement si aucune HNO n'a encore de n° de change) :
     ALTER TABLE dbo.saisie_activite DROP CONSTRAINT CK_saisie_astreinte_champs;
     ALTER TABLE dbo.saisie_activite WITH CHECK ADD CONSTRAINT CK_saisie_astreinte_champs CHECK (
       (rubrique = 'ASTREINTE' AND change_ou_motif IS NOT NULL AND LEN(TRIM(change_ou_motif)) > 0 AND est_resolu IS NOT NULL)
       OR (rubrique <> 'ASTREINTE' AND change_ou_motif IS NULL AND est_resolu IS NULL AND resolution_commentaire IS NULL));
     DELETE FROM dbo.migration_journal WHERE script_nom = N'017_hno_numero_change';
   ============================================================================ */

SET NOCOUNT ON;
SET XACT_ABORT ON;
GO

BEGIN TRANSACTION;

IF EXISTS (SELECT 1 FROM sys.check_constraints
           WHERE name = N'CK_saisie_astreinte_champs' AND parent_object_id = OBJECT_ID(N'dbo.saisie_activite'))
BEGIN
    ALTER TABLE dbo.saisie_activite DROP CONSTRAINT CK_saisie_astreinte_champs;
END;

ALTER TABLE dbo.saisie_activite WITH CHECK
    ADD CONSTRAINT CK_saisie_astreinte_champs CHECK
    (
        (rubrique = 'ASTREINTE'
            AND change_ou_motif IS NOT NULL
            AND LEN(TRIM(change_ou_motif)) > 0
            AND est_resolu IS NOT NULL)
        OR
        (rubrique = 'HNO'
            AND (change_ou_motif IS NULL OR LEN(TRIM(change_ou_motif)) > 0)
            AND est_resolu IS NULL
            AND resolution_commentaire IS NULL)
        OR
        (rubrique NOT IN ('ASTREINTE', 'HNO')
            AND change_ou_motif IS NULL
            AND est_resolu IS NULL
            AND resolution_commentaire IS NULL)
    );

COMMIT TRANSACTION;
GO

IF OBJECT_ID(N'dbo.migration_journal', N'U') IS NOT NULL
BEGIN
    IF NOT EXISTS (SELECT 1 FROM dbo.migration_journal WHERE script_nom = N'017_hno_numero_change')
    BEGIN
        INSERT INTO dbo.migration_journal (script_nom) VALUES (N'017_hno_numero_change');
    END;
END;
PRINT N'Migration 017 appliquée : n° de change autorisé sur les HNO (saisie_activite.change_ou_motif).';
GO
