/* ============================================================================
   Migration 016 - Remise en ordre des heures de début / fin (HNO et astreintes)
   Demande client du 31/10/2026 : les heures vont dans heure_debut / heure_fin,
   JAMAIS dans la description.

   Entre le 30/10 et le 31/10/2026, une version du site a écrit les heures en
   tête de description (« [HH:MM-HH:MM] texte »). Ce script, à passer UNE fois :
     1. supprime le déclencheur TR_saisie_heures_description s'il a été créé
        (version précédente de ce script) ;
     2. pour chaque ligne HNO / ASTREINTE dont la description commence par
        « [HH:MM-HH:MM] » : copie les heures dans heure_debut / heure_fin
        (si elles sont vides) et retire le préfixe de la description.
   Tout se fait dans une transaction : en cas d'erreur, rien n'est modifié.
   Les lignes sans préfixe ne sont pas touchées. Une ligne dont la description
   ne contiendrait QUE le préfixe garde sa description (elle ne peut pas être
   vide : CK_saisie_description_non_vide) ; elle est comptée à part.

   Prérequis : migration 012 appliquée (colonnes heure_debut / heure_fin).
   Idempotent (réexécutable sans dégât).
     sqlcmd -S .\SQLEXPRESS -d cra -b -i 016_heures_hno_depuis_description.sql

   Contrôle après passage (doit renvoyer 0) :
     SELECT COUNT(*) FROM dbo.saisie_activite
     WHERE rubrique IN ('HNO','ASTREINTE') AND description LIKE N'[[][0-9][0-9]:[0-9][0-9]-%';
   ============================================================================ */

SET NOCOUNT ON;
SET XACT_ABORT ON;
GO

IF COL_LENGTH(N'dbo.saisie_activite', N'heure_debut') IS NULL
   OR COL_LENGTH(N'dbo.saisie_activite', N'heure_fin') IS NULL
BEGIN
    THROW 51016, N'Colonnes heure_debut / heure_fin absentes : appliquer d''abord la migration 012 - abandon.', 1;
END;
GO

/* Étape 1 - déclencheur de la version précédente : retiré ------------------ */
IF OBJECT_ID(N'dbo.TR_saisie_heures_description', N'TR') IS NOT NULL
BEGIN
    DROP TRIGGER dbo.TR_saisie_heures_description;
    PRINT N'Déclencheur TR_saisie_heures_description supprimé.';
END;
GO

/* Étape 2 - heures remises dans leurs colonnes, description nettoyée -------- */
BEGIN TRANSACTION;

DECLARE @a_traiter TABLE (id BIGINT PRIMARY KEY, debut TIME(0) NOT NULL, fin TIME(0) NOT NULL, reste NVARCHAR(500) NOT NULL);

/* Préfixe attendu, positions fixes : [ H H : M M - H H : M M ]
                                      1 2 . . . 6 7 8 . . 12 13 */
INSERT INTO @a_traiter (id, debut, fin, reste)
SELECT s.id, h.debut, h.fin, LTRIM(SUBSTRING(s.description, 14, 500))
  FROM dbo.saisie_activite AS s
  CROSS APPLY (SELECT TRY_CONVERT(TIME(0), SUBSTRING(s.description, 2, 5)) AS debut,
                      TRY_CONVERT(TIME(0), SUBSTRING(s.description, 8, 5)) AS fin) AS h
 WHERE s.rubrique IN ('HNO', 'ASTREINTE')
   AND LEN(s.description) >= 13
   AND SUBSTRING(s.description, 1, 1)  = N'['
   AND SUBSTRING(s.description, 4, 1)  = N':'
   AND SUBSTRING(s.description, 7, 1)  = N'-'
   AND SUBSTRING(s.description, 10, 1) = N':'
   AND SUBSTRING(s.description, 13, 1) = N']'
   AND h.debut IS NOT NULL
   AND h.fin IS NOT NULL
   AND h.debut <> h.fin;

UPDATE s
   SET heure_debut = CASE WHEN s.heure_debut IS NULL AND s.heure_fin IS NULL THEN t.debut ELSE s.heure_debut END,
       heure_fin   = CASE WHEN s.heure_debut IS NULL AND s.heure_fin IS NULL THEN t.fin   ELSE s.heure_fin   END,
       description = CASE WHEN LEN(t.reste) > 0 THEN t.reste ELSE s.description END
  FROM dbo.saisie_activite AS s
  JOIN @a_traiter AS t ON t.id = s.id;

DECLARE @total INT = (SELECT COUNT(*) FROM @a_traiter);
DECLARE @vides INT = (SELECT COUNT(*) FROM @a_traiter WHERE LEN(reste) = 0);

COMMIT TRANSACTION;

PRINT CONCAT(N'Lignes remises en ordre (heures dans heure_debut / heure_fin, description nettoyée) : ', @total - @vides);
IF @vides > 0
    PRINT CONCAT(N'Lignes dont la description ne contenait que les heures (heures copiées, description laissée telle quelle) : ', @vides);
GO

/* Étape 3 - journal de migration (si présent) ------------------------------ */
IF OBJECT_ID(N'dbo.migration_journal', N'U') IS NOT NULL
BEGIN
    IF NOT EXISTS (SELECT 1 FROM dbo.migration_journal WHERE script_nom = N'016_heures_hno_depuis_description')
    BEGIN
        INSERT INTO dbo.migration_journal (script_nom) VALUES (N'016_heures_hno_depuis_description');
    END;
END;
GO
